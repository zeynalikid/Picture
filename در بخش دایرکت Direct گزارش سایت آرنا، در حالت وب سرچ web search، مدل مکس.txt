#### کد نمونه Webhook Handler (FastAPI روی Vercel)

```python
# api/webhook/instagram.py
from fastapi import FastAPI, Request, HTTPException
from supabase import create_client
import hashlib
import os

app = FastAPI()
supabase = create_client(os.getenv("SUPABASE_URL"), os.getenv("SUPABASE_KEY"))

SPAM_KEYWORDS = ["follow for follow", "f4f", "spam_keyword"]
MAX_MSGS_PER_USER_PER_DAY = 5

@app.post("/api/webhook/instagram")
async def instagram_webhook(request: Request):
    # ۱. تأیید امضای Webhook
    body = await request.body()
    signature = request.headers.get("X-Webhook-Signature", "")
    expected = hashlib.sha256(
        f"{os.getenv('WEBHOOK_SECRET')}{body.decode()}".encode()
    ).hexdigest()
    if signature != f"sha256={expected}":
        raise HTTPException(status_code=401, detail="Invalid signature")

    data = await request.json()
    
    # ۲. استخراج اطلاعات پیام
    event_type = data.get("event")  # "comment.received" or "dm.received"
    sender_id = data.get("sender", {}).get("id")
    message_text = data.get("text", "")
    
    # ۳. لایه فیلتر اسپم
    if await is_spam(sender_id, message_text):
        return {"status": "filtered", "reason": "spam"}
    
    # ۴. بررسی محدودیت ماهانه ارسال
    monthly_count = await get_monthly_send_count()
    if monthly_count >= 9500:  # بافر ۵۰۰ تایی قبل از ۱۰k
        # فقط ذخیره می‌کنیم، ارسال نمی‌کنیم
        await save_to_queue(sender_id, message_text, event_type, priority="deferred")
        return {"status": "queued_deferred"}
    
    # ۵. ذخیره در صف
    await save_to_queue(sender_id, message_text, event_type, priority="normal")
    
    return {"status": "queued"}

async def is_spam(sender_id: str, text: str) -> bool:
    # بررسی بلاک‌لیست
    result = supabase.table("blocked_senders").select("id").eq("sender_id", sender_id).execute()
    if result.data:
        return True
    
    # بررسی کلمات اسپم
    text_lower = text.lower()
    if any(keyword in text_lower for keyword in SPAM_KEYWORDS):
        return True
    
    # بررسی تعداد پیام در روز (جلوگیری از flood)
    from datetime import datetime, timedelta
    today = datetime.now().date().isoformat()
    result = supabase.table("message_queue")\
        .select("id")\
        .eq("sender_id", sender_id)\
        .gte("created_at", today)\
        .execute()
    if len(result.data) >= MAX_MSGS_PER_USER_PER_DAY:
        return True
    
    return False

async def get_monthly_send_count() -> int:
    from datetime import datetime
    month_start = datetime.now().replace(day=1).date().isoformat()
    result = supabase.table("message_queue")\
        .select("id")\
        .eq("status", "processed")\
        .gte("created_at", month_start)\
        .execute()
    return len(result.data)

async def save_to_queue(sender_id, text, event_type, priority="normal"):
    supabase.table("message_queue").insert({
        "sender_id": sender_id,
        "message_text": text,
        "event_type": event_type,
        "status": "pending",
        "priority": priority
    }).execute()
```

#### کد نمونه Queue Processor (Vercel Cron)

```python
# api/process-queue.py
import os
import httpx
from supabase import create_client
import google.generativeai as genai

supabase = create_client(os.getenv("SUPABASE_URL"), os.getenv("SUPABASE_KEY"))
genai.configure(api_key=os.getenv("GEMINI_API_KEY"))

SYSTEM_PROMPT = """
تو دستیار برند کودک @zeynalikids هستی.
- پاسخ‌ها باید مودبانه، کوتاه (زیر ۱۵۰ کاراکتر) و به فارسی باشند.
- از ایموجی مناسب استفاده کن.
- اگر سوال درباره قیمت است، بگو 'لطفاً DM بدید'.
- هرگز اطلاعات شخصی نده.
"""

async def process_queue():
    # ۱. دریافت پیام‌های pending (batch 10 تایی)
    result = supabase.table("message_queue")\
        .select("*")\
        .eq("status", "pending")\
        .order("created_at")\
        .limit(10)\
        .execute()
    
    messages = result.data
    if not messages:
        return {"processed": 0}
    
    model = genai.GenerativeModel("gemini-1.5-flash")
    
    for msg in messages:
        try:
            # ۲. تولید پاسخ با Gemini
            response = model.generate_content(
                f"{SYSTEM_PROMPT}\n\nپیام کاربر: {msg['message_text']}"
            )
            reply_text = response.text.strip()
            
            # ۳. ارسال پاسخ از طریق SocialAPI.ai یا Zernio
            await send_reply(msg, reply_text)
            
            # ۴. بروزرسانی وضعیت
            supabase.table("message_queue")\
                .update({"status": "processed", "reply_text": reply_text})\
                .eq("id", msg["id"])\
                .execute()
                
        except Exception as e:
            supabase.table("message_queue")\
                .update({"status": "error", "error_message": str(e)})\
                .eq("id", msg["id"])\
                .execute()
    
    return {"processed": len(messages)}

async def send_reply(msg: dict, reply_text: str):
    """
    استراتژی دوگانه:
    - اگر مصرف ماهانه < 10k: از Zernio استفاده می‌کنیم (رایگان‌تر)
    - اگر مصرف ماهانه > 9500: به SocialAPI.ai برمی‌گردیم (۵۰ رایگان)
    """
    monthly_count = await get_monthly_send_count()
    
    if monthly_count < 9500:
        # Zernio Inbox API
        async with httpx.AsyncClient() as client:
            if msg["event_type"] == "comment.received":
                await client.post(
                    "https://api.zernio.com/v1/comments/reply",
                    headers={"Authorization": f"Bearer {os.getenv('ZERNIO_API_KEY')}"},
                    json={
                        "comment_id": msg["platform_message_id"],
                        "text": reply_text
                    }
                )
            elif msg["event_type"] == "dm.received":
                await client.post(
                    "https://api.zernio.com/v1/messages/send",
                    headers={"Authorization": f"Bearer {os.getenv('ZERNIO_API_KEY')}"},
                    json={
                        "conversation_id": msg["conversation_id"],
                        "text": reply_text
                    }
                )
    else:
        # SocialAPI.ai (backup)
        async with httpx.AsyncClient() as client:
            await client.post(
                "https://api.social-api.ai/v1/inbox/reply",
                headers={"Authorization": f"Bearer {os.getenv('SOCIALAPI_KEY')}"},
                json={
                    "interaction_id": msg["platform_message_id"],
                    "text": reply_text
                }
            )
```

### ❓ سوال ۴: مدیریت صف برای زیر ۱۰,۰۰۰ پیام/ماه

#### استراتژی‌های کاهش مصرف

**حساب کتاب واقعی:**
```
۵۰۰ پیام/روز × ۳۰ روز = ۱۵,۰۰۰ پیام/ماه (بدون فیلتر)
هدف: زیر ۱۰,۰۰۰ پیام/ماه = حذف ۳۳% پیام‌ها
```

**لایه‌های فیلتر (به ترتیب اثربخشی):**

| لایه | تکنیک | کاهش تخمینی | پیاده‌سازی |
|-------|--------|-------------|-----------|
| **۱. بلاک‌لیست** | بلاک کردن سندرهای مکرر بی‌ربط | ۱۵-۲۰% | جدول `blocked_senders` |
| **۲. Dedup** | بررسی پیام‌های یکسان ظرف ۵ دقیقه | ۵-۱۰% | Hash پیام + timestamp |
| **۳. Rate Limit** | حداکثر ۳ پاسخ به هر کاربر/روز | ۱۰-۱۵% | شمارنده روزانه در Supabase |
| **۴. Keyword Filter** | رد پیام‌های spam/emoji-only/کمتر از ۵ کاراکتر | ۱۰-۱۵% | Regex |
| **۵. Priority Queue** | فقط به پیام‌های با keyword مهم پاسخ دهید | انتخابی | امتیازدهی |

**قابلیت‌های Zernio برای کمک:**

Zernio از طریق API امکان خواندن پیام‌ها را می‌دهد. فیلترینگ هوشمند باید در لایه Vercel شما انجام شود چون Zernio خودش ابزار بلاک داخلی منتشرشده‌ای برای این کار ارائه نمی‌دهد. اما Zernio برای هر event از طریق Webhook به شما اطلاع می‌دهد و شما می‌توانید قبل از پاسخ دادن، فیلتر کنید.

**SQL Schema پیشنهادی در Supabase:**

```sql
-- جدول صف پیام‌ها
CREATE TABLE message_queue (
    id UUID DEFAULT gen_random_uuid() PRIMARY KEY,
    sender_id TEXT NOT NULL,
    message_text TEXT NOT NULL,
    message_hash TEXT,  -- برای deduplication
    event_type TEXT,    -- 'comment.received' or 'dm.received'
    platform_message_id TEXT,
    conversation_id TEXT,
    status TEXT DEFAULT 'pending', -- pending/processed/spam/blocked/deferred
    priority TEXT DEFAULT 'normal', -- high/normal/low/deferred
    reply_text TEXT,
    error_message TEXT,
    created_at TIMESTAMPTZ DEFAULT NOW(),
    processed_at TIMESTAMPTZ
);

-- جدول بلاک‌لیست
CREATE TABLE blocked_senders (
    id UUID DEFAULT gen_random_uuid() PRIMARY KEY,
    sender_id TEXT UNIQUE NOT NULL,
    reason TEXT,
    blocked_at TIMESTAMPTZ DEFAULT NOW()
);

-- جدول آمار ماهانه (برای کنترل ۱۰k)
CREATE TABLE monthly_stats (
    id UUID DEFAULT gen_random_uuid() PRIMARY KEY,
    year_month TEXT NOT NULL,  -- '2026-10'
    messages_sent INTEGER DEFAULT 0,
    messages_filtered INTEGER DEFAULT 0,
    updated_at TIMESTAMPTZ DEFAULT NOW()
);

-- Index برای کارایی
CREATE INDEX idx_queue_status ON message_queue(status, created_at);
CREATE INDEX idx_queue_sender ON message_queue(sender_id, created_at);
CREATE INDEX idx_queue_hash ON message_queue(message_hash);

-- تابع deduplication
CREATE OR REPLACE FUNCTION is_duplicate_message(
    p_sender_id TEXT,
    p_message_hash TEXT
) RETURNS BOOLEAN AS $$
BEGIN
    RETURN EXISTS (
        SELECT 1 FROM message_queue
        WHERE sender_id = p_sender_id
        AND message_hash = p_message_hash
        AND created_at > NOW() - INTERVAL '10 minutes'
    );
END;
$$ LANGUAGE plpgsql;
```

    """
    
    async with httpx.AsyncClient() as client:
        await client.post(
            f"https://api.telegram.org/bot{bot_token}/sendMessage",
            json={
                "chat_id": admin_chat_id,

            }
        )

## خلاصه پاسخ‌های مستقیم به سوالات شما:

**۱. جایگزین ChatbotX:**
بهترین گزینه **SocialAPI.ai Hobby Plan** است — رایگان، بدون کارت اعتباری، با همان پلتفرم‌ها و قابلیت‌های نسخه‌های پولی. خواندن پیام‌ها (DM، کامنت) کاملاً نامحدود و رایگان است. تنها ارسال پاسخ‌ها به ۵۰ تعامل/ماه محدود است.

**۲. اتصال بدون اپ متا:**
از طریق Zernio می‌توانید کامنت‌ها را بخوانید، پاسخ دهید، پنهان کنید و مدیریت کنید — هر event کامنت از طریق Normalized Webhook ارسال می‌شود و نیازی به polling ندارید.

**۳. وضعیت Ayrshare:**
در ۲۰۲۶، Ayrshare چهار پلن دارد: Premium $149/ماه، Launch $299/ماه، Business $599/ماه، و Enterprise. هیچ پلن رایگان دائمی وجود ندارد، فقط ۲۸ روز trial.

**۴. وضعیت Publer:**
Publer هیچ Social Inbox ندارد — مدیریت کامنت/DM از Publer ممکن نیست. API نیز فقط در Business plan ($21+/ماه) در دسترس است.

**۵. کنترل زیر ۱۰k پیام:**
طبق مستندات Zernio در میانه ۲۰۲۶، ارسال Inbox بعد از ۱۰,۰۰۰ پیام/ماه به صورت متری اضافه می‌شود. با مدیریت صف هوشمند در Supabase و لایه‌های فیلتر اسپم، می‌توان مصرف را کنترل کرد.
